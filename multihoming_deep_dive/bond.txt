#Create bond interface
ip link add dev bond0 type bond
#Use LACP on bond interface
echo 802.3ad > /sys/class/net/bond0/bonding/mode
echo fast > /sys/class/net/bond0/bonding/lacp_rate
echo layer3+4 > /sys/class/net/bond0/bonding/xmit_hash_policy
#Assign physical interfaces to bond
ip link set dev eth1 down
ip link set dev eth2 down
ip link set dev eth1 master bond0
ip link set dev eth2 master bond0
ip link set dev bond0 up
ip link set dev eth1 up
ip link set dev eth2 up
#Create sub-interfaces with VLANs
ip link add link bond0 name bond0.10 type vlan id 10
ip link add link bond0 name bond0.20 type vlan id 20
ip a a 10.10.10.10/24 dev bond0.10
ip a a 10.20.20.10/24 dev bond0.20
ip link set dev bond0.10 up
ip link set dev bond0.20 up
